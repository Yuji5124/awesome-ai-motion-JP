const s = { p: 0 };
tl.to(s, { p: 1, duration: 0.6, ease: 'none', onUpdate: () => {
  const k = Math.sin(Math.PI * s.p);
  gsap.set(['.prev', '.next'], { filter: `invert(${k * 0.85}) contrast(${1 + 0.4 * k}) saturate(${1 + 0.2 * k})` });
} }, 0);
tl.set('.prev', { autoAlpha: 0 }, 0.3).set('.next', { autoAlpha: 1 }, 0.3);
