const N = 8, T = 6;
boxes.forEach((b, i) => {
  const ph = (i / N) * Math.PI * 2;
  tl.to(b, { duration: T, ease: 'none', onUpdate() {
    const a = ph + this.progress() * Math.PI * 2;
    gsap.set(b, { x: Math.sin(a) * 520, z: Math.cos(a) * 260, y: Math.sin(a * 2) * 80, rotationY: -Math.sin(a) * 40 });
  } }, 0);
});
