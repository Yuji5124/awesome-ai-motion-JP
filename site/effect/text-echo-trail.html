for (let k = 0; k <= 6; k++) {
  const layer = k === 0 ? main : main.cloneNode(true);
  if (k) { layer.style.opacity = 0.5 * (1 - k / 7); stage.appendChild(layer); }
  tl.fromTo(layer, {x:-900}, {x:0, duration:0.9, ease:'power3.out'}, 0.2 + k * 0.045);
  if (k) tl.set(layer, {opacity:0}, 1.3);
}
