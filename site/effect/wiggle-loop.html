const tl = gsap.timeline({ paused: true });
tl.to(state, { t: 3, duration: 3, ease: 'none', onUpdate: () => {
  const a = state.t * Math.PI * 2 / 3;
  gsap.set('.wiggle', { x: 5 * Math.sin(3*a) + 3 * Math.sin(5*a), rotation: 2 * Math.sin(2*a) });
} });
